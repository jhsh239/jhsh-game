    const firebaseConfig = {

      measurementId: "G-X1PDRENPHW"
    };

    firebase.initializeApp(firebaseConfig);
    const db = firebase.database();
    const tugRef = db.ref('tug_of_war/red_percent');
    const danmakuRef = db.ref('tug_of_war/messages');

    const percentRedEl = document.getElementById('percentRed');
    const percentWhiteEl = document.getElementById('percentWhite');
    const barRedEl = document.getElementById('barRed');
    const barWhiteEl = document.getElementById('barWhite');
    const statusBadge = document.getElementById('statusBadge');
    const winnerModal = document.getElementById('winnerModal');
    const winnerText = document.getElementById('winnerText');
    const commentInput = document.getElementById('commentInput');
    const danmakuContainer = document.getElementById('danmaku-container');

    let currentRed = 50;
    let hasTriggeredWin = false;

    // 1. 即時監聽比分拉鋸
    tugRef.on('value', (snap) => {
      let val = snap.val();
      if (val === null) {
        tugRef.set(50);
        val = 50;
      }

      currentRed = val;
      const currentWhite = 100 - currentRed;

      percentRedEl.textContent = currentRed;
      percentWhiteEl.textContent = currentWhite;
      barRedEl.style.width = currentRed + '%';
      barWhiteEl.style.width = currentWhite + '%';

      if (currentRed > 65) {
        statusBadge.textContent = '🏮 紅隊攤位熱評如潮！全面領先！';
      } else if (currentRed < 35) {
        statusBadge.textContent = '🍢 白隊攤位香氣爆棚！大獲好評！';
      } else {
        statusBadge.textContent = '⚔️ 兩隊人氣難分軒輊！快留言力挺！';
      }

      if (currentRed >= 100) {
        showWinner('🚩 紅隊攤位獲勝！全場最好吃！');
      } else if (currentRed <= 0) {
        showWinner('🍲 白隊攤位獲勝！全場最美味！');
      } else {
        winnerModal.style.display = 'none';
        hasTriggeredWin = false;
      }
    });

    // 2. 監聽全場新留言，飄出彈幕
    danmakuRef.limitToLast(1).on('child_added', (snapshot) => {
      const msg = snapshot.val();
      if (msg && msg.text) {
        spawnDanmaku(msg.text, msg.team);
      }
    });

    function spawnDanmaku(text, team) {
      const bubble = document.createElement('div');
      bubble.className = `danmaku-item ${team === 'red' ? 'danmaku-red' : 'danmaku-white'}`;
      bubble.textContent = `${team === 'red' ? '🔴' : '⚪'} ${text}`;
      
      // 隨機水平位置（留出邊距，避免貼邊）
      const randomLeft = 10 + Math.random() * 60;
      bubble.style.left = `${randomLeft}%`;

      danmakuContainer.appendChild(bubble);

      setTimeout(() => {
        bubble.remove();
      }, 4500);
    }

    // 3. 送出留言 + 加分
    function submitVote(team) {
      const text = commentInput.value.trim();
      if (!text) {
        alert('請先輸入心得或推薦文字再投票喔！');
        commentInput.focus();
        return;
      }

      if (currentRed <= 0 || currentRed >= 100) return;

      // 推播留言到 Firebase
      danmakuRef.push({
        text: text,
        team: team,
        timestamp: Date.now()
      });

      // 拔河拉鋸數值 +1 或 -1
      const power = team === 'red' ? +1 : -1;
      tugRef.transaction((val) => {
        if (val === null) return 50;
        let next = val + power;
        if (next > 100) next = 100;
        if (next < 0) next = 0;
        return next;
      });

      // 清空輸入框
      commentInput.value = '';
    }

    document.getElementById('btnRed').addEventListener('click', () => submitVote('red'));
    document.getElementById('btnWhite').addEventListener('click', () => submitVote('white'));

    function showWinner(title) {
      winnerText.textContent = title;
      winnerModal.style.display = 'flex';
      if (!hasTriggeredWin) {
        confetti({ particleCount: 150, spread: 80, origin: { y: 0.6 } });
        hasTriggeredWin = true;
      }
    }

    document.getElementById('btnRestart').addEventListener('click', () => {
      tugRef.set(50);
      danmakuRef.remove(); // 重設時清空留言歷史
    });
